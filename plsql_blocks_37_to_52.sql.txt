-- 39. Copy column data type to a variable using %TYPE
DECLARE
    v_name Doctor.doc_name%TYPE;
    v_sal Doctor.salary%TYPE;
BEGIN
    SELECT doc_name, salary INTO v_name, v_sal FROM Doctor WHERE doc_id = 101;
    DBMS_OUTPUT.PUT_LINE('Doctor: ' || v_name || ' | Salary: ' || v_sal);
END;
/

-- 40. Fetch an entire row into a variable using %ROWTYPE

END;
/

-- 41. Use IF-ELSIF-ELSE conditions in PL/SQL
DECLARE
    v_salary NUMBER := 75000;
BEGIN
    IF v_salary >= 80000 THEN
        DBMS_OUTPUT.PUT_LINE('Category: Senior Specialist');
    ELSIF v_salary >= 60000 THEN
        DBMS_OUTPUT.PUT_LINE('Category: Mid-Level Specialist');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Category: Junior Doctor');
    END IF;
END;
/

-- 44. Run a basic LOOP with EXIT WHEN condition in PL/SQL
DECLARE
    i NUMBER := 1;
BEGIN
    LOOP
        DBMS_OUTPUT.PUT_LINE('Count: ' || i);
        i := i + 1;
        EXIT WHEN i > 3;
    END LOOP;
END;
/

-- 45. Run a WHILE loop in PL/SQL
DECLARE
    i NUMBER := 1;
BEGIN

        DBMS_OUTPUT.PUT_LINE('Iteration: ' || i);

    END LOOP;
END;
/

-- 47. Fetch multiple rows using an explicit CURSOR in PL/SQL
DECLARE
    CURSOR c_doctor IS 
        SELECT doc_name, salary FROM Doctor;
    v_name Doctor.doc_name%TYPE;
    v_sal  Doctor.salary%TYPE;
BEGIN
    OPEN c_doctor;
    LOOP
        FETCH c_doctor INTO v_name, v_sal;
        EXIT WHEN c_doctor%NOTFOUND;
        DBMS_OUTPUT.PUT_LINE('Doctor Name: ' || v_name || ' | Salary: ' || v_sal);
    END LOOP;
    CLOSE c_doctor;
END;
/

-- 48. Catch divide-by-zero error using ZERO_DIVIDE exception
DECLARE
    num NUMBER;
BEGIN
    num := 100 / 0;
EXCEPTION
    WHEN ZERO_DIVIDE THEN
        DBMS_OUTPUT.PUT_LINE('Error: Cannot divide by zero!');
END;
/
